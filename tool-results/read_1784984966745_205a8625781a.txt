   121→bool IsAdministrator() {
   122→    BOOL isAdmin = FALSE;
   123→    PSID adminSid = NULL;
   124→    SID_IDENTIFIER_AUTHORITY ntAuthority = SECURITY_NTAuthority;
   125→    
   126→    if (AllocateAndInitializeSid(&ntAuthority, 2, SECURITY_BUILTIN_DOMAIN_RID, 
   127→        DOMAIN_ALIAS_RID_ADMINS, 0, 0, 0, 0, 0, 0, &adminSid)) {
   128→        if (!CheckTokenMembership(NULL, adminSid, &isAdmin)) {
   129→            isAdmin = FALSE;
   130→        }
   131→        FreeSid(adminSid);
   132→    }
   133→    
   134→    return isAdmin == TRUE;
   135→}

   151→std::set<std::wstring> GetCurrentUSBDrives() {
   152→    std::set<std::wstring> drives;
   153→    DWORD drivesMask = GetLogicalDrives();
   154→    
   155→    for (char drive = 'A'; drive <= 'Z'; drive++) {
   156→        if (drivesMask & (1 << (drive - 'A'))) {
   157→            std::wstring path = std::wstring(1, drive) + L":\\";
   158→            UINT type = GetDriveType(path.c_str());
   159→            
   160→            if (type == DRIVE_REMOVABLE || type == DRIVE_FIXED) {
   161→                drives.insert(std::wstring(1, drive) + L":");
   162→            }
   163→        }
   164→    }
   165→    
   166→    return drives;
   167→}

   169→bool IsUSBDeviceConnected(const std::wstring& hardwareId) {
   170→    HDEVINFO hDevInfo = SetupDiGetClassDevs(NULL, L"USB", NULL, DIGCF_PRESENT | DIGCF_ALLCLASSES);
   171→    if (hDevInfo == INVALID_HANDLE_VALUE) return false;
   172→    
   173→    bool found = false;
   174→    SP_DEVINFO_DATA devInfo = {};
   175→    devInfo.cbSize = sizeof(SP_DEVINFO_DATA);
   176→    
   177→    for (DWORD i = 0; SetupDiEnumDeviceInfo(hDevInfo, i, &devInfo); i++) {
   178→        wchar_t deviceId[MAX_DEVICE_ID_LEN];
   179→        if (SetupDiGetDeviceRegistryPropertyW(hDevInfo, &devInfo, SPDRP_HARDWAREID,
   180→            NULL, (PBYTE)deviceId, sizeof(deviceId), NULL)) {
   181→            
   182→            std::wstring idStr(deviceId);
   183→            if (idStr.find(hardwareId) != std::wstring::npos) {
   184→                found = true;
   185→                break;
   186→            }
   187→        }
   188→    }
   189→    
   190→    SetupDiDestroyDeviceInfoList(hDevInfo);
   191→    return found;
   192→}

   194→bool DeleteRegistryKeyRecursive(HKEY hRootKey, const std::wstring& subKey) {
   195→    HKEY hKey;
   196→    LONG result = RegOpenKeyExW(hRootKey, subKey.c_str(), 0, KEY_READ | KEY_WRITE | KEY_WOW64_64KEY, &hKey);
   197→    
   198→    if (result != ERROR_SUCCESS) return false;
   199→    
   200→    DWORD subKeysCount = 0, maxSubKeyLen = 0;
   201→    RegQueryInfoKeyW(hKey, NULL, NULL, NULL, &subKeysCount, &maxSubKeyLen, NULL,
   202→        NULL, NULL, NULL, NULL, NULL);
   203→    
   204→    if (subKeysCount > 0 && maxSubKeyLen > 0) {
   205→        std::vector<wchar_t> subKeyName(maxSubKeyLen + 1);
   206→        
   207→        for (DWORD i = subKeysCount; i > 0; i--) {
   208→            DWORD nameSize = maxSubKeyLen + 1;
   209→            FILETIME lastWriteTime;
   210→            
   211→            if (RegEnumKeyExW(hKey, i - 1, subKeyName.data(), &nameSize,
   212→                NULL, NULL, NULL, &lastWriteTime) == ERROR_SUCCESS) {
   213→                
   214→                std::wstring childPath = subKey + L"\\" + subKeyName.data();
   215→                DeleteRegistryKeyRecursive(hRootKey, childPath);
   216→            }
   217→        }
   218→    }
   219→    
   220→    RegCloseKey(hKey);
   221→    
   222→    result = RegDeleteKeyExW(hRootKey, subKey.c_str(), KEY_WOW64_64KEY, 0);
   223→    return (result == ERROR_SUCCESS || result == ERROR_FILE_NOT_FOUND);
   224→}

   226→OperationResult CleanUSBHistory(std::function<void(int)> progressCallback) {
   227→    OperationResult result{true, 0, "", {}};
   228→    
   229→    logger.log(L"Начало очистки истории USB-устройств...");
   230→    
   231→    HKEY hUsbStorKey;
   232→    LONG openResult = RegOpenKeyExW(HKEY_LOCAL_MACHINE, USB_STOR_KEY, 0, 
   233→        KEY_READ | KEY_WRITE | KEY_WOW64_64KEY, &hUsbStorKey);
   234→    
   235→    if (openResult != ERROR_SUCCESS) {
   236→        result.success = false;
   237→        result.message = "Ошибка открытия реестра USBSTOR";
   238→        return result;
   239→    }
   240→    
   241→    // Получаем список устройств
   242→    DWORD subKeysCount = 0, maxSubKeyLen = 0;
   243→    RegQueryInfoKeyW(hUsbStorKey, NULL, NULL, NULL, &subKeysCount,
   244→        &maxSubKeyLen, NULL, NULL, NULL, NULL, NULL, NULL);
   245→    
   246→    int processed = 0;
   247→    
   248→    if (subKeysCount > 0 && maxSubKeyLen > 0) {
   249→        std::vector<wchar_t> deviceClassName(maxSubKeyLen + 1);
   250→        
   251→        for (DWORD i = 0; i < subKeysCount; i++) {
   252→            DWORD nameSize = maxSubKeyLen + 1;
   253→            FILETIME lastWriteTime;
   254→            
   255→            if (RegEnumKeyExW(hUsbStorKey, i, deviceClassName.data(), &nameSize,
   256→                NULL, NULL, NULL, &lastWriteTime) == ERROR_SUCCESS) {
   257→                
   258→                std::wstring classPath = std::wstring(USB_STOR_KEY) + L"\\" + deviceClassName.data();
   259→                
   260→                HKEY hClassKey;
   261→                if (RegOpenKeyExW(HKEY_LOCAL_MACHINE, classPath.c_str(), 0,
   262→                    KEY_READ | KEY_WRITE | KEY_WOW64_64KEY, &hClassKey) == ERROR_SUCCESS) {
   263→                    
   264→                    DWORD instanceCount = 0, maxInstanceLen = 0;
   265→                    RegQueryInfoKeyW(hClassKey, NULL, NULL, NULL, &instanceCount,
   266→                        &maxInstanceLen, NULL, NULL, NULL, NULL, NULL, NULL);
   267→                    
   268→                    if (instanceCount > 0 && maxInstanceLen > 0) {
   269→                        std::vector<wchar_t> instanceName(maxInstanceLen + 1);
   270→                        
   271→                        for (DWORD j = 0; j < instanceCount; j++) {
   272→                            DWORD instNameSize = maxInstanceLen + 1;
   273→                            
   274→                            if (RegEnumKeyExW(hClassKey, j, instanceName.data(), &instNameSize,
   275→                                NULL, NULL, NULL, &lastWriteTime) == ERROR_SUCCESS) {
   276→                                
   277→                                std::wstring fullDeviceId = deviceClassName.data();
   278→                                fullDeviceId += L"&";
   279→                                fullDeviceId += instanceName.data();
   280→                                
   281→                                if (!IsUSBDeviceConnected(fullDeviceId)) {
   282→                                    std::wstring instancePath = classPath + L"\\" + instanceName.data();
   283→                                    
   284→                                    logger.log(L"Удаление: " + fullDeviceId);
   285→                                    
   286→                                    if (DeleteRegistryKeyRecursive(HKEY_LOCAL_MACHINE, instancePath)) {
   287→                                        result.itemsProcessed++;
   288→                                        result.details.push_back(L"Удалено: " + fullDeviceId);
   289→                                    } else {
   290→                                        result.details.push_back(L"Ошибка удаления: " + fullDeviceId);
   291→                                    }
   292→                                } else {
   293→                                    result.details.push_back(L"Пропущено (подключено): " + fullDeviceId);
   294→                                }
   295→                                
   296→                                processed++;
   297→                                if (progressCallback) {
   298→                                    progressCallback((processed * 100) / (int)(subKeysCount * 2));
   299→                                }
   300→                            }
   301→                        }
   302→                    }
   303→                    
   304→                    RegCloseKey(hClassKey);
   305→                }
   306→                
   307→                if (progressCallback) {
   308→                    progressCallback((processed * 100) / (int)(subKeysCount * 2));
   309→                }
   310→            }
   311→        }
   312→    }
   313→    
   314→    RegCloseKey(hUsbStorKey);
   315→    
   316→    // Очищаем MountedDevices
   317→    HKEY hMountedKey;
   318→    if (RegOpenKeyExW(HKEY_LOCAL_MACHINE, MOUNTED_DEVICES_KEY, 0,
   319→        KEY_READ | KEY_WRITE | KEY_WOW64_64KEY, &hMountedKey) == ERROR_SUCCESS) {
   320→        
   321→        DWORD valuesCount = 0, maxValueNameLen = 0;
   322→        RegQueryInfoKeyW(hMountedKey, NULL, NULL, NULL, NULL, NULL, NULL,
   323→            &valuesCount, &maxValueNameLen, NULL, NULL, NULL);
   324→        
   325→        if (valuesCount > 0 && maxValueNameLen > 0) {
   326→            std::vector<wchar_t> valueName(maxValueNameLen + 1);
   327→            
   328→            for (DWORD i = 0; i < valuesCount; i++) {
   329→                DWORD valueNameSize = maxValueNameLen + 1;
   330→                
   331→                if (RegEnumValueW(hMountedKey, i, valueName.data(), &valueNameSize,
   332→                    NULL, NULL, NULL, NULL) == ERROR_SUCCESS) {
   333→                    
   334→                    std::wstring valName(valueName.data());
   335→                    
   336→                    if (valName.find(L"\\DosDevices\\") != std::wstring::npos ||
   337→                        valName.find(L"\\??\\") != std::wstring::npos) {
   338→                        
   339→                        std::wstring driveLetter;
   340→                        size_t pos = valName.find(L"\\DosDevices\\");
   341→                        if (pos != std::wstring::npos) {
   342→                            driveLetter = valName.substr(pos + 12);
   343→                        } else {
   344→                            pos = valName.find(L"\\??\\");
   345→                            if (pos != std::wstring::npos) {
   346→                                driveLetter = valName.substr(pos + 4);
   347→                            }
   348→                        }
   349→                        
   350→                        if (!driveLetter.empty()) {
   351→                            UINT type = GetDriveType(driveLetter.c_str());
   352→                            if (type == DRIVE_NO_ROOT_DIR || type == DRIVE_UNKNOWN) {
   353→                                RegDeleteValueW(hMountedKey, valName.c_str());
   354→                                result.itemsProcessed++;
   355→                                processed++;
   356→                            }
   357→                        }
   358→                    }
   359→                }
   360→                
   361→                if (progressCallback) {
   362→                    progressCallback((processed * 100) / (int)(subKeysCount * 2));
   363→                }
   364→            }
   365→        }
   366→        
   367→        RegCloseKey(hMountedKey);
   368→    }
   369→    
   370→    result.message = "Очистка завершена успешно";
   371→    logger.log(L"Завершено. Удалено: " + std::to_wstring(result.itemsProcessed));
   372→    
   373→    return result;
   374→}

   376→int RemoveZoneIdentifiers(const std::wstring& rootPath, std::function<void(int)> progressCallback, int totalFiles) {
   377→    static int processed = 0;
   378→    int removedCount = 0;
   379→    
   380→    WIN32_FIND_DATAW findData;
   381→    std::wstring searchPattern = rootPath + L"*.*";
   382→    
   383→    HANDLE hFind = FindFirstFileW(searchPattern.c_str(), &findData);
   384→    if (hFind == INVALID_HANDLE_VALUE) return 0;
   385→    
   386→    do {
   387→        if (wcscmp(findData.cFileName, L".") == 0 || wcscmp(findData.cFileName, L"..") == 0)
   388→            continue;
   389→        
   390→        std::wstring fullPath = rootPath + findData.cFileName;
   391→        
   392→        if (findData.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) {
   393→            fullPath += L"\\";
   394→            removedCount += RemoveZoneIdentifiers(fullPath, progressCallback, totalFiles);
   395→        } else {
   396→            std::wstring zoneFile = fullPath + L":Zone.Identifier";
   397→            DWORD attrs = GetFileAttributesW(zoneFile.c_str());
   398→            
   399→            if (attrs != INVALID_FILE_ATTRIBUTES) {
   400→                SetFileAttributesW(zoneFile.c_str(), FILE_ATTRIBUTE_NORMAL);
   401→                
   402→                if (DeleteFileW(zoneFile.c_str())) {
   403→                    removedCount++;
   404→                    logger.log(L"Удален Zone.Identifier: " + fullPath);
   405→                }
   406→            }
   407→            
   408→            processed++;
   409→            if (progressCallback && totalFiles > 0) {
   410→                progressCallback((processed * 100) / totalFiles);
   411→            }
   412→        }
   413→        
   414→    } while (FindNextFileW(hFind, &findData));
   415→    
   416→    FindClose(hFind);
   417→    return removedCount;
   418→}

   420→OperationResult CleanFlashLabels(std::function<void(int)> progressCallback) {
   421→    OperationResult result{true, 0, "", {}};
   422→    
   423→    logger.log(L"Начало очистки меток флешек...");
   424→    
   425→    int totalRemoved = 0;
   426→    std::set<std::wstring> processedDrives;
   427→    
   428→    DWORD drivesMask = GetLogicalDrives();
   429→    int driveCount = 0;
   430→    
   431→    // Сначала считаем диски
   432→    for (char drive = 'A'; drive <= 'Z'; drive++) {
   433→        if (drivesMask & (1 << (drive - 'A'))) {
   434→            std::wstring drivePath = std::wstring(1, drive) + L":\\";
   435→            UINT type = GetDriveType(drivePath.c_str());
   436→            if (type == DRIVE_REMOVABLE || type == DRIVE_FIXED) {
   437→                driveCount++;
   438→            }
   439→        }
   440→    }
   441→    
   442→    int currentDrive = 0;
   443→    
   444→    for (char drive = 'A'; drive <= 'Z'; drive++) {
   445→        if (!(drivesMask & (1 << (drive - 'A')))) continue;
   446→        
   447→        std::wstring drivePath = std::wstring(1, drive) + L":\\";
   448→        UINT type = GetDriveType(drivePath.c_str());
   449→        
   450→        if (type == DRIVE_REMOVABLE || type == DRIVE_FIXED) {
   451→            currentDrive++;
   452→            
   453→            wchar_t fileSystem[MAX_PATH];
   454→            if (GetVolumeInformationW(drivePath.c_str(), NULL, 0, NULL, NULL, NULL, 
   455→                fileSystem, MAX_PATH)) {
   456→                
   457→                std::wstring driveLetter(1, drive);
   458→                driveLetter += L":";
   459→                
   460→                if (processedDrives.find(driveLetter) == processedDrives.end()) {
   461→                    processedDrives.insert(driveLetter);
   462→                    
   463→                    logger.log(L"Обработка диска: " + driveLetter);
   464→                    
   465→                    int removed = RemoveZoneIdentifiers(drivePath, 
   466→                        [&, currentDrive](int p) {
   467→                            if (progressCallback) {
   468→                                progressCallback(((currentDrive - 1) * 100 + p) / driveCount);
   469→                            }
   470→                        }, 0);
   471→                    
   472→                    totalRemoved += removed;
   473→                    result.details.push_back(L"Диск " + driveLetter + L": удалено " + 
   474→                        std::to_wstring(removed) + L" меток");
   475→                }
   476→            }
   477→            
   478→            if (progressCallback) {
   479→                progressCallback((currentDrive * 100) / driveCount);
   480→            }
   481→        }
   482→    }
   483→    
   484→    SHChangeNotify(SHCNE_UPDATEDIR, SHCNF_PATH, NULL, NULL);
   485→    
   486→    result.itemsProcessed = totalRemoved;
   487→    result.message = "Очистка меток завершена успешно";
   488→    logger.log(L"Завершено. Удалено меток: " + std::to_wstring(totalRemoved));
   489→    
   490→    return result;
   491→}

   782→class SystemTrayIcon {
   783→private:
   784→    NOTIFYICONDATA nid = {};
   785→    HWND hWnd = NULL;
   786→    
   787→public:
   788→    bool Create(HWND parentWnd, const wchar_t* tooltip, HICON icon) {
   789→        hWnd = parentWnd;
   790→        
   791→        nid.cbSize = sizeof(NOTIFYICONDATA);
   792→        nid.hWnd = hWnd;
   793→        nid.uID = 1;
   794→        nid.uFlags = NIF_ICON | NIF_MESSAGE | NIF_TIP;
   795→        nid.uCallbackMessage = WM_APP + 1;
   796→        nid.hIcon = icon ? icon : LoadIcon(NULL, IDI_WINLOGO);
   797→        wcscpy_s(nid.szTip, tooltip);
   798→        
   799→        return Shell_NotifyIcon(NIM_ADD, &nid) != FALSE;
   800→    }
   801→    
   802→    void Remove() {
   803→        Shell_NotifyIcon(NIM_DELETE, &nid);
   804→    }
   805→    
   806→    void ShowBalloon(const wchar_t* title, const wchar_t* text, int timeout = 3000) {
   807→        wcscpy_s(nid.szInfoTitle, title);
   808→        wcscpy_s(nid.szInfo, text);
   809→        nid.uTimeout = timeout;
   810→        nid.uFlags |= NIF_INFO;
   811→        nid.dwInfoFlags = NIIF_INFO;
   812→        Shell_NotifyIcon(NIM_MODIFY, &nid);
   813→    }
   814→    
   815→    void SetTooltip(const wchar_t* tooltip) {
   816→        wcscpy_s(nid.szTip, tooltip);
   817→        Shell_NotifyIcon(NIM_MODIFY, &nid);
   818→    }
   819→};
