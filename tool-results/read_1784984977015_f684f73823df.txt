   121→   121→bool IsAdministrator() {
   122→   122→    BOOL isAdmin = FALSE;
   123→   123→    PSID adminSid = NULL;
   124→   124→    SID_IDENTIFIER_AUTHORITY ntAuthority = SECURITY_NTAuthority;
   125→   125→    
   126→   126→    if (AllocateAndInitializeSid(&ntAuthority, 2, SECURITY_BUILTIN_DOMAIN_RID, 
   127→   127→        DOMAIN_ALIAS_RID_ADMINS, 0, 0, 0, 0, 0, 0, &adminSid)) {
   128→   128→        if (!CheckTokenMembership(NULL, adminSid, &isAdmin)) {
   129→   129→            isAdmin = FALSE;
   130→   130→        }
   131→   131→        FreeSid(adminSid);
   132→   132→    }
   133→   133→    
   134→   134→    return isAdmin == TRUE;

   151→   151→std::set<std::wstring> GetCurrentUSBDrives() {
   152→   152→    std::set<std::wstring> drives;
   153→   153→    DWORD drivesMask = GetLogicalDrives();
   154→   154→    
   155→   155→    for (char drive = 'A'; drive <= 'Z'; drive++) {
   156→   156→        if (drivesMask & (1 << (drive - 'A'))) {
   157→   157→            std::wstring path = std::wstring(1, drive) + L":\\";
   158→   158→            UINT type = GetDriveType(path.c_str());
   159→   159→            
   160→   160→            if (type == DRIVE_REMOVABLE || type == DRIVE_FIXED) {
   161→   161→                drives.insert(std::wstring(1, drive) + L":");
   162→   162→            }
   163→   163→        }
   164→   164→    }
   165→   165→    
   166→   166→    return drives;

   169→   169→bool IsUSBDeviceConnected(const std::wstring& hardwareId) {
   170→   170→    HDEVINFO hDevInfo = SetupDiGetClassDevs(NULL, L"USB", NULL, DIGCF_PRESENT | DIGCF_ALLCLASSES);
   171→   171→    if (hDevInfo == INVALID_HANDLE_VALUE) return false;
   172→   172→    
   173→   173→    bool found = false;
   174→   174→    SP_DEVINFO_DATA devInfo = {};
   175→   175→    devInfo.cbSize = sizeof(SP_DEVINFO_DATA);
   176→   176→    
   177→   177→    for (DWORD i = 0; SetupDiEnumDeviceInfo(hDevInfo, i, &devInfo); i++) {
   178→   178→        wchar_t deviceId[MAX_DEVICE_ID_LEN];
   179→   179→        if (SetupDiGetDeviceRegistryPropertyW(hDevInfo, &devInfo, SPDRP_HARDWAREID,
   180→   180→            NULL, (PBYTE)deviceId, sizeof(deviceId), NULL)) {
   181→   181→            
   182→   182→            std::wstring idStr(deviceId);
   183→   183→            if (idStr.find(hardwareId) != std::wstring::npos) {
   184→   184→                found = true;
   185→   185→                break;
   186→   186→            }
   187→   187→        }
   188→   188→    }
   189→   189→    
   190→   190→    SetupDiDestroyDeviceInfoList(hDevInfo);
   191→   191→    return found;
   192→   192→}

   194→   194→bool DeleteRegistryKeyRecursive(HKEY hRootKey, const std::wstring& subKey) {
   195→   195→    HKEY hKey;
   196→   196→    LONG result = RegOpenKeyExW(hRootKey, subKey.c_str(), 0, KEY_READ | KEY_WRITE | KEY_WOW64_64KEY, &hKey);
   197→   197→    
   198→   198→    if (result != ERROR_SUCCESS) return false;
   199→   199→    
   200→   200→    DWORD subKeysCount = 0, maxSubKeyLen = 0;
   201→   201→    RegQueryInfoKeyW(hKey, NULL, NULL, NULL, &subKeysCount, &maxSubKeyLen, NULL,
   202→   202→        NULL, NULL, NULL, NULL, NULL);
   203→   203→    
   204→   204→    if (subKeysCount > 0 && maxSubKeyLen > 0) {
   205→   205→        std::vector<wchar_t> subKeyName(maxSubKeyLen + 1);
   206→   206→        
   207→   207→        for (DWORD i = subKeysCount; i > 0; i--) {
   208→   208→            DWORD nameSize = maxSubKeyLen + 1;
   209→   209→            FILETIME lastWriteTime;
   210→   210→            
   211→   211→            if (RegEnumKeyExW(hKey, i - 1, subKeyName.data(), &nameSize,
   212→   212→                NULL, NULL, NULL, &lastWriteTime) == ERROR_SUCCESS) {
   213→   213→                
   214→   214→                std::wstring childPath = subKey + L"\\" + subKeyName.data();
   215→   215→                DeleteRegistryKeyRecursive(hRootKey, childPath);
   216→   216→            }
   217→   217→        }
   218→   218→    }
   219→   219→    
   220→   220→    RegCloseKey(hKey);
   221→   221→    
   222→   222→    result = RegDeleteKeyExW(hRootKey, subKey.c_str(), KEY_WOW64_64KEY, 0);
   223→   223→    return (result == ERROR_SUCCESS || result == ERROR_FILE_NOT_FOUND);
   224→   224→}

   226→   226→OperationResult CleanUSBHistory(std::function<void(int)> progressCallback) {
   227→   227→    OperationResult result{true, 0, "", {}};
   228→   228→    
   229→   229→    logger.log(L"Начало очистки истории USB-устройств...");
   230→   230→    
   231→   231→    HKEY hUsbStorKey;
   232→   232→    LONG openResult = RegOpenKeyExW(HKEY_LOCAL_MACHINE, USB_STOR_KEY, 0, 
   233→   233→        KEY_READ | KEY_WRITE | KEY_WOW64_64KEY, &hUsbStorKey);
   234→   234→    
   235→   235→    if (openResult != ERROR_SUCCESS) {
   236→   236→        result.success = false;
   237→   237→        result.message = "Ошибка открытия реестра USBSTOR";
   238→   238→        return result;
   239→   239→    }
   240→   240→    
   241→   241→    // Получаем список устройств
   242→   242→    DWORD subKeysCount = 0, maxSubKeyLen = 0;
   243→   243→    RegQueryInfoKeyW(hUsbStorKey, NULL, NULL, NULL, &subKeysCount,
   244→   244→        &maxSubKeyLen, NULL, NULL, NULL, NULL, NULL, NULL);
   245→   245→    
   246→   246→    int processed = 0;
   247→   247→    
   248→   248→    if (subKeysCount > 0 && maxSubKeyLen > 0) {
   249→   249→        std::vector<wchar_t> deviceClassName(maxSubKeyLen + 1);
   250→   250→        
   251→   251→        for (DWORD i = 0; i < subKeysCount; i++) {
   252→   252→            DWORD nameSize = maxSubKeyLen + 1;
   253→   253→            FILETIME lastWriteTime;
   254→   254→            
   255→   255→            if (RegEnumKeyExW(hUsbStorKey, i, deviceClassName.data(), &nameSize,
   256→   256→                NULL, NULL, NULL, &lastWriteTime) == ERROR_SUCCESS) {
   257→   257→                
   258→   258→                std::wstring classPath = std::wstring(USB_STOR_KEY) + L"\\" + deviceClassName.data();
   259→   259→                
   260→   260→                HKEY hClassKey;
   261→   261→                if (RegOpenKeyExW(HKEY_LOCAL_MACHINE, classPath.c_str(), 0,
   262→   262→                    KEY_READ | KEY_WRITE | KEY_WOW64_64KEY, &hClassKey) == ERROR_SUCCESS) {
   263→   263→                    
   264→   264→                    DWORD instanceCount = 0, maxInstanceLen = 0;
   265→   265→                    RegQueryInfoKeyW(hClassKey, NULL, NULL, NULL, &instanceCount,
   266→   266→                        &maxInstanceLen, NULL, NULL, NULL, NULL, NULL, NULL);
   267→   267→                    
   268→   268→                    if (instanceCount > 0 && maxInstanceLen > 0) {
   269→   269→                        std::vector<wchar_t> instanceName(maxInstanceLen + 1);
   270→   270→                        
   271→   271→                        for (DWORD j = 0; j < instanceCount; j++) {
   272→   272→                            DWORD instNameSize = maxInstanceLen + 1;
   273→   273→                            
   274→   274→                            if (RegEnumKeyExW(hClassKey, j, instanceName.data(), &instNameSize,
   275→   275→                                NULL, NULL, NULL, &lastWriteTime) == ERROR_SUCCESS) {
   276→   276→                                
   277→   277→                                std::wstring fullDeviceId = deviceClassName.data();
   278→   278→                                fullDeviceId += L"&";
   279→   279→                                fullDeviceId += instanceName.data();
   280→   280→                                
   281→   281→                                if (!IsUSBDeviceConnected(fullDeviceId)) {
   282→   282→                                    std::wstring instancePath = classPath + L"\\" + instanceName.data();
   283→   283→                                    
   284→   284→                                    logger.log(L"Удаление: " + fullDeviceId);
   285→   285→                                    
   286→   286→                                    if (DeleteRegistryKeyRecursive(HKEY_LOCAL_MACHINE, instancePath)) {
   287→   287→                                        result.itemsProcessed++;
   288→   288→                                        result.details.push_back(L"Удалено: " + fullDeviceId);
   289→   289→                                    } else {
   290→   290→                                        result.details.push_back(L"Ошибка удаления: " + fullDeviceId);
   291→   291→                                    }
   292→   292→                                } else {
   293→   293→                                    result.details.push_back(L"Пропущено (подключено): " + fullDeviceId);
   294→   294→                                }
   295→   295→                                
   296→   296→                                processed++;
   297→   297→                                if (progressCallback) {
   298→   298→                                    progressCallback((processed * 100) / (int)(subKeysCount * 2));
   299→   299→                                }
   300→   300→                            }
   301→   301→                        }
   302→   302→                    }
   303→   303→                    
   304→   304→                    RegCloseKey(hClassKey);
   305→   305→                }
   306→   306→                
   307→   307→                if (progressCallback) {
   308→   308→                    progressCallback((processed * 100) / (int)(subKeysCount * 2));
   309→   309→                }
   310→   310→            }
   311→   311→        }
   312→   312→    }
   313→   313→    
   314→   314→    RegCloseKey(hUsbStorKey);
   315→   315→    
   316→   316→    // Очищаем MountedDevices
   317→   317→    HKEY hMountedKey;
   318→   318→    if (RegOpenKeyExW(HKEY_LOCAL_MACHINE, MOUNTED_DEVICES_KEY, 0,
   319→   319→        KEY_READ | KEY_WRITE | KEY_WOW64_64KEY, &hMountedKey) == ERROR_SUCCESS) {
   320→   320→        
   321→   321→        DWORD valuesCount = 0, maxValueNameLen = 0;
   322→   322→        RegQueryInfoKeyW(hMountedKey, NULL, NULL, NULL, NULL, NULL, NULL,
   323→   323→            &valuesCount, &maxValueNameLen, NULL, NULL, NULL);
   324→   324→        
   325→   325→        if (valuesCount > 0 && maxValueNameLen > 0) {
   326→   326→            std::vector<wchar_t> valueName(maxValueNameLen + 1);
   327→   327→            
   328→   328→            for (DWORD i = 0; i < valuesCount; i++) {
   329→   329→                DWORD valueNameSize = maxValueNameLen + 1;
   330→   330→                
   331→   331→                if (RegEnumValueW(hMountedKey, i, valueName.data(), &valueNameSize,
   332→   332→                    NULL, NULL, NULL, NULL) == ERROR_SUCCESS) {
   333→   333→                    
   334→   334→                    std::wstring valName(valueName.data());
   335→   335→                    
   336→   336→                    if (valName.find(L"\\DosDevices\\") != std::wstring::npos ||
   337→   337→                        valName.find(L"\\??\\") != std::wstring::npos) {
   338→   338→                        
   339→   339→                        std::wstring driveLetter;
   340→   340→                        size_t pos = valName.find(L"\\DosDevices\\");
   341→   341→                        if (pos != std::wstring::npos) {
   342→   342→                            driveLetter = valName.substr(pos + 12);
   343→   343→                        } else {
   344→   344→                            pos = valName.find(L"\\??\\");
   345→   345→                            if (pos != std::wstring::npos) {
   346→   346→                                driveLetter = valName.substr(pos + 4);
   347→   347→                            }
   348→   348→                        }
   349→   349→                        
   350→   350→                        if (!driveLetter.empty()) {
   351→   351→                            UINT type = GetDriveType(driveLetter.c_str());
   352→   352→                            if (type == DRIVE_NO_ROOT_DIR || type == DRIVE_UNKNOWN) {
   353→   353→                                RegDeleteValueW(hMountedKey, valName.c_str());
   354→   354→                                result.itemsProcessed++;
   355→   355→                                processed++;
   356→   356→                            }
   357→   357→                        }
   358→   358→                    }
   359→   359→                }
   360→   360→                
   361→   361→                if (progressCallback) {
   362→   362→                    progressCallback((processed * 100) / (int)(subKeysCount * 2));
   363→   363→                }
   364→   364→            }
   365→   365→        }
   366→   366→        
   367→   367→        RegCloseKey(hMountedKey);
   368→   368→    }
   369→   369→    
   370→   370→    result.message = "Очистка завершена успешно";
   371→   371→    logger.log(L"Завершено. Удалено: " + std::to_wstring(result.itemsProcessed));
   372→   372→    
   373→   373→    return result;
   374→   374→}

   376→   376→int RemoveZoneIdentifiers(const std::wstring& rootPath, std::function<void(int)> progressCallback, int totalFiles) {
   377→   377→    static int processed = 0;
   378→   378→    int removedCount = 0;
   379→   379→    
   380→   380→    WIN32_FIND_DATAW findData;
   381→   381→    std::wstring searchPattern = rootPath + L"*.*";
   382→   382→    
   383→   383→    HANDLE hFind = FindFirstFileW(searchPattern.c_str(), &findData);
   384→   384→    if (hFind == INVALID_HANDLE_VALUE) return 0;
   385→   385→    
   386→   386→    do {
   387→   387→        if (wcscmp(findData.cFileName, L".") == 0 || wcscmp(findData.cFileName, L"..") == 0)
   388→   388→            continue;
   389→   389→        
   390→   390→        std::wstring fullPath = rootPath + findData.cFileName;
   391→   391→        
   392→   392→        if (findData.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) {
   393→   393→            fullPath += L"\\";
   394→   394→            removedCount += RemoveZoneIdentifiers(fullPath, progressCallback, totalFiles);
   395→   395→        } else {
   396→   396→            std::wstring zoneFile = fullPath + L":Zone.Identifier";
   397→   397→            DWORD attrs = GetFileAttributesW(zoneFile.c_str());
   398→   398→            
   399→   399→            if (attrs != INVALID_FILE_ATTRIBUTES) {
   400→   400→                SetFileAttributesW(zoneFile.c_str(), FILE_ATTRIBUTE_NORMAL);
   401→   401→                
   402→   402→                if (DeleteFileW(zoneFile.c_str())) {
   403→   403→                    removedCount++;
   404→   404→                    logger.log(L"Удален Zone.Identifier: " + fullPath);
   405→   405→                }
   406→   406→            }
   407→   407→            
   408→   408→            processed++;
   409→   409→            if (progressCallback && totalFiles > 0) {
   410→   410→                progressCallback((processed * 100) / totalFiles);
   411→   411→            }
   412→   412→        }
   413→   413→        
   414→   414→    } while (FindNextFileW(hFind, &findData));
   415→   415→    
   416→   416→    FindClose(hFind);
   417→   417→    return removedCount;
   418→   418→}

   451→   451→            currentDrive++;
   452→   452→            
   453→   453→            wchar_t fileSystem[MAX_PATH];
   454→   454→            if (GetVolumeInformationW(drivePath.c_str(), NULL, 0, NULL, NULL, NULL, 
   455→   455→                fileSystem, MAX_PATH)) {
   456→   456→                
   457→   457→                std::wstring driveLetter(1, drive);
   458→   458→                driveLetter += L":";
   459→   459→                
   460→   460→                if (processedDrives.find(driveLetter) == processedDrives.end()) {
   461→   461→                    processedDrives.insert(driveLetter);
   462→   462→                    
   463→   463→                    logger.log(L"Обработка диска: " + driveLetter);
   464→   464→                    
   465→   465→                    int removed = RemoveZoneIdentifiers(drivePath, 
   466→   466→                        [&, currentDrive](int p) {
   467→   467→                            if (progressCallback) {
   468→   468→                                progressCallback(((currentDrive - 1) * 100 + p) / driveCount);
   469→   469→                            }
   470→   470→                        }, 0);
   471→   471→                    
   472→   472→                    totalRemoved += removed;
   473→   473→                    result.details.push_back(L"Диск " + driveLetter + L": удалено " + 
   474→   474→                        std::to_wstring(removed) + L" меток");
   475→   475→                }
   476→   476→            }
   477→   477→            
   478→   478→            if (progressCallback) {
   479→   479→                progressCallback((currentDrive * 100) / driveCount);
   480→   480→            }
   481→   481→        }
   482→   482→    }
   483→   483→    
   484→   484→    SHChangeNotify(SHCNE_UPDATEDIR, SHCNF_PATH, NULL, NULL);
   485→   485→    
   486→   486→    result.itemsProcessed = totalRemoved;
   487→   487→    result.message = "Очистка меток завершена успешно";
   488→   488→    logger.log(L"Завершено. Удалено меток: " + std::to_wstring(totalRemoved));
   489→   489→    
   490→   490→    return result;
   491→   491→}
